import React, { useState, useEffect } from "react";
import { IoMdCut } from "react-icons/io";
import { BsFillLightningChargeFill } from "react-icons/bs";

function BookingCard({ text, price, text2, price2, setService }) {
  const [bookingOptions, setBookingOptions] = useState(null);

  useEffect(() => {
    // console.log("bookingOptions changed:", bookingOptions);
  }, [bookingOptions]);

  const isSelected = (key) => bookingOptions === key;

  const baseCard =
    "relative rounded-2xl border bg-white/5 backdrop-blur-md " +
    "border-white/15 hover:border-[#DDCA7D]/40 transition-all duration-300 " +
    "shadow-sm hover:shadow-lg overflow-hidden";

  const selectedRing =
    "ring-2 ring-[#DDCA7D]/40 shadow-[0_0_40px_-10px_rgba(221,202,125,0.55)]";

  const pillIcon =
    "mx-auto mb-4 p-4 w-20 h-20 rounded-full bg-[#DDCA7D]/15 flex items-center justify-center";

  const titleClass = "text-2xl sm:text-3xl font-semibold leading-none tracking-tight text-[#DDCA7D]";
  const priceClass = "text-xl sm:text-2xl font-bold text-[#DDCA7D]";
  const container = "grid grid-cols-1 md:grid-cols-2 gap-4 md:gap-6 w-full max-w-5xl mx-auto mt-8 px-4";

  return (
    <div className={container}>
      {/* Card 1 – Low Taper */}
      <div
        role="button"
        tabIndex={0}
        aria-pressed={isSelected("booking-card-1")}
        onClick={() => { setBookingOptions("booking-card-1"); setService("Low Taper"); }}
        onKeyDown={(e) => { if (e.key === "Enter" || e.key === " ") { setBookingOptions("booking-card-1"); setService("Low Taper"); }}}
        className={`${baseCard} ${isSelected("booking-card-1") ? selectedRing : ""}`}
      >
        {/* subtle gradient sheen */}
        <div className="pointer-events-none absolute inset-0 bg-gradient-to-br from-[#DDCA7D]/8 via-transparent to-transparent" />
        {/* check badge when selected */}
        {isSelected("booking-card-1") && (
          <div className="absolute top-3 right-3 w-7 h-7 bg-[#DDCA7D] text-black rounded-full grid place-items-center">
            <svg className="w-4 h-4" viewBox="0 0 20 20" fill="currentColor">
              <path
                fillRule="evenodd"
                d="M16.707 5.293a1 1 0 010 1.414l-8 8a1 1 0 01-1.414 0l-4-4a1 1 0 011.414-1.414L8 12.586l7.293-7.293a1 1 0 011.414 0z"
                clipRule="evenodd"
              />
            </svg>
          </div>
        )}

        <div className="h-full min-h-[14rem] sm:min-h-[16rem] flex flex-col items-center justify-center gap-3 px-6 py-8 text-center">
          <div className={pillIcon}>
            <IoMdCut className={`w-full h-full ${isSelected("booking-card-1") ? "text-[#DDCA7D]" : "text-white/80"}`} />
          </div>
          <h3 className={titleClass}>{text}</h3>
          <div className={priceClass}>{price}</div>

          {/* footer hint */}
          <div className="mt-2 text-white/60 text-xs">
            Click to select
          </div>
        </div>
      </div>

      {/* Card 2 – Emergency Cut */}
      <div
        role="button"
        tabIndex={0}
        aria-pressed={isSelected("booking-card-2")}
        onClick={() => { setBookingOptions("booking-card-2"); setService("Emergency Cut"); }}
        onKeyDown={(e) => { if (e.key === "Enter" || e.key === " ") { setBookingOptions("booking-card-2"); setService("Emergency Cut"); }}}
        className={`${baseCard} ${isSelected("booking-card-2") ? selectedRing : ""}`}
      >
        {/* subtle gradient sheen */}
        <div className="pointer-events-none absolute inset-0 bg-gradient-to-br from-[#DDCA7D]/8 via-transparent to-transparent" />
        {/* check badge when selected */}
        {isSelected("booking-card-2") && (
          <div className="absolute top-3 right-3 w-7 h-7 bg-[#DDCA7D] text-black rounded-full grid place-items-center">
            <svg className="w-4 h-4" viewBox="0 0 20 20" fill="currentColor">
              <path
                fillRule="evenodd"
                d="M16.707 5.293a1 1 0 010 1.414l-8 8a1 1 0 01-1.414 0l-4-4a1 1 0 011.414-1.414L8 12.586l7.293-7.293a1 1 0 011.414 0z"
                clipRule="evenodd"
              />
            </svg>
          </div>
        )}

        <div className="h-full min-h-[14rem] sm:min-h-[16rem] flex flex-col items-center justify-center gap-3 px-6 py-8 text-center">
          <div className={pillIcon}>
            <BsFillLightningChargeFill className={`w-full h-full ${isSelected("booking-card-2") ? "text-[#DDCA7D]" : "text-white/80"}`} />
          </div>
          <h3 className={titleClass}>{text2}</h3>
          <div className={priceClass}>{price2}</div>

          {/* footer hint */}
          <div className="mt-2 text-white/60 text-xs">
            Priority scheduling
          </div>
        </div>
      </div>
    </div>
  );
}

export default BookingCard;
